```bash
git clone https://github.com/arfinmahmudshiblu/End-to-End-Wine-Quality-Prediction.git
```

```bash
conda create -n cvdev python=3.11 -y
```

```bash
conda activate cvdev
```

```bash
pip install -r requirements.txt
```
